name: Deploy Astro site to Pages

on:
  push:
    branches: [main]
  
  workflow_dispatch:

permissions:
  contents: write
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: false

env:
  BUILD_PATH: "."

jobs:
  build:
    name: Build
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v7

      - name: Detect package manager
        id: detect-package-manager
        run: |
          if [ -f "${{ github.workspace }}/yarn.lock" ]; then
            echo "manager=yarn" >> $GITHUB_OUTPUT
            echo "command=install" >> $GITHUB_OUTPUT
            echo "runner=yarn" >> $GITHUB_OUTPUT
            echo "lockfile=yarn.lock" >> $GITHUB_OUTPUT
            exit 0
          elif [ -f "${{ github.workspace }}/package.json" ]; then
            echo "manager=npm" >> $GITHUB_OUTPUT
            echo "command=ci" >> $GITHUB_OUTPUT
            echo "runner=npx --no-install" >> $GITHUB_OUTPUT
            echo "lockfile=package-lock.json" >> $GITHUB_OUTPUT
            exit 0
          else
            echo "Unable to determine package manager"
            exit 1
          fi

      - name: Setup Node
        uses: actions/setup-node@v7
        with:
          node-version: "24"
          cache: ${{ steps.detect-package-manager.outputs.manager }}
          cache-dependency-path: ${{ env.BUILD_PATH }}/${{ steps.detect-package-manager.outputs.lockfile }}

      - name: Setup Pages
        id: pages
        uses: actions/configure-pages@v6

      - name: Install dependencies
        run: ${{ steps.detect-package-manager.outputs.manager }} ${{ steps.detect-package-manager.outputs.command }}
        working-directory: ${{ env.BUILD_PATH }}

      - name: Build with Astro
        run: |
          ${{ steps.detect-package-manager.outputs.runner }} astro build \
            --site "${{ steps.pages.outputs.origin }}" \
            --base "${{ steps.pages.outputs.base_path }}"
        working-directory: ${{ env.BUILD_PATH }}

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v5
        with:
          path: ${{ env.BUILD_PATH }}/dist

      - name: Deploy to gh-pages branch
        uses: peaceiris/actions-gh-pages@v4
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ${{ env.BUILD_PATH }}/dist
          publish_branch: gh-pages

  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    needs: build
    runs-on: ubuntu-latest
    name: Deploy
    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v5

name: Repo Tree

on:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  tree:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Build tree
        run: |
          OUTPUT="repo-tree.txt"
          {
            echo "=== REPO TREE ==="
            echo "Generated: $(date -u)"
            echo "Commit: $(git rev-parse --short HEAD)"
            echo ""

            echo "=== STRUCTURE ==="
            if command -v tree >/dev/null 2>&1; then
              tree -a -I 'node_modules|.git|dist|.wrangler|.astro|.cache|.next|coverage|.DS_Store|package-lock.json' --dirsfirst
            else
              find . -type d \
                -not -path "./.git*" \
                -not -path "./node_modules*" \
                -not -path "./dist*" \
                -not -path "./.wrangler*" \
                -not -path "./.astro*" \
                -not -path "./.cache*" \
                | sort
            fi
            echo ""

            echo "=== FILE LIST (with size & lines) ==="
            printf "%-60s %10s %10s\n" "FILE" "SIZE" "LINES"
            printf "%-60s %10s %10s\n" "----" "----" "-----"

            find . -type f \
              -not -path "./.git/*" \
              -not -path "./node_modules/*" \
              -not -path "./dist/*" \
              -not -path "./.wrangler/*" \
              -not -path "./.astro/*" \
              -not -path "./.cache/*" \
              -not -name "repo-tree.txt" \
              -not -name "package-lock.json" \
              -not -name "*.log" \
              -not -name ".DS_Store" \
              | sort | while read -r file; do
                size=$(du -h "$file" 2>/dev/null | cut -f1)
                if file "$file" | grep -q text; then
                  lines=$(wc -l < "$file" 2>/dev/null || echo "0")
                else
                  lines="-"
                fi
                printf "%-60s %10s %10s\n" "$file" "$size" "$lines"
              done

            echo ""

            echo "=== FILE COUNT BY TYPE ==="
            find . -type f \
              -not -path "./.git/*" \
              -not -path "./node_modules/*" \
              -not -path "./dist/*" \
              -not -path "./.wrangler/*" \
              -not -path "./.astro/*" \
              -not -name "repo-tree.txt" \
              -not -name "package-lock.json" \
              | sed 's/.*\.//' | sort | uniq -c | sort -rn

            echo ""
            echo "=== DIRECTORY SIZES ==="
            du -sh --exclude='.git' --exclude='node_modules' --exclude='dist' --exclude='.wrangler' --exclude='.astro' ./* 2>/dev/null | sort -rh | head -30

          } > "$OUTPUT"

          echo "Generated $(wc -l < $OUTPUT) lines"

      - name: Commit
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          git add repo-tree.txt

const currentPath = Astro.url.pathname;

function isActive(href: string): boolean {
  if (href === '/') return currentPath === '/';
  const matches = currentPath === href || currentPath.startsWith(href);
  if (!matches) return false;
  const hasMoreSpecific = NAV_LINKS.some(({ href: otherHref }) => {
    if (otherHref === href) return false;
    if (otherHref === '/') return false;
    if (otherHref.length <= href.length) return false;
    return currentPath === otherHref || currentPath.startsWith(otherHref);
  });
  return !hasMoreSpecific;
}

import Icon from '~/user-components/Icons.astro';

interface Props {
  current: number;
  total: number;
  base: string;
}

const { current, total, base } = Astro.props;

if (total <= 1) return null;

function urlFor(page: number): string {
  return page === 1 ? `${base}/` : `${base}/${page}/`;
}

function buildPages(): (number | '...')[] {
  if (total <= 7) {
    return Array.from({ length: total }, (_, i) => i + 1);
  }

  const pages: (number | '...')[] = [1];

  if (current > 3) pages.push('...');

  const start = Math.max(2, current - 1);
  const end = Math.min(total - 1, current + 1);

  for (let i = start; i <= end; i++) pages.push(i);

  if (current < total - 2) pages.push('...');
  pages.push(total);

  return pages;
}

const pages = buildPages();

<script>
  const iframe = document.getElementById('player-iframe') as HTMLIFrameElement | null;
  const currentLabel = document.getElementById('stream-current');
  const radios = document.querySelectorAll<HTMLInputElement>('input[name="stream"]');

  function selectStream(radio: HTMLInputElement) {
    if (!iframe || !radio) return;

    iframe.src = radio.value;

    if (currentLabel) {
      currentLabel.textContent = `${radio.dataset.quality} · ${radio.dataset.server}`;
    }

    localStorage.setItem('qimochi:stream', radio.id);
  }

  radios.forEach((radio) => {
    radio.addEventListener('change', () => {
      if (radio.checked) selectStream(radio);
    });
  });

  const saved = localStorage.getItem('qimochi:stream');
  if (saved) {
    const savedRadio = document.getElementById(saved) as HTMLInputElement | null;
    if (savedRadio) {
      savedRadio.checked = true;
      selectStream(savedRadio);
    }
  }
</script>

function isActive(href: string): boolean {
  if (href === '/') return currentPath === '/';
  const matches = currentPath === href || currentPath.startsWith(href);
  if (!matches) return false;
  const hasMoreSpecific = NAV_LINKS.some(({ href: otherHref }) => {
    if (otherHref === href) return false;
    if (otherHref === '/') return false;
    if (otherHref.length <= href.length) return false;

  });
  return !hasMoreSpecific;
}

import Icon from '~/user-components/Icons.astro';

interface Props {
  current: number;
  total: number;
  base: string;
}

const { current, total, base } = Astro.props;

if (total <= 1) return null;

function urlFor(page: number): string {
  return page === 1 ? `${base}/` : `${base}/${page}/`;
}

function buildPages(): (number | '...')[] {
  if (total <= 7) {
    return Array.from({ length: total }, (_, i) => i + 1);
  }

  const pages: (number | '...')[] = [1];

  if (current > 3) pages.push('...');

  const start = Math.max(2, current - 1);
  const end = Math.min(total - 1, current + 1);

  for (let i = start; i <= end; i++) pages.push(i);

  if (current < total - 2) pages.push('...');
  pages.push(total);

  return pages;
}

const pages = buildPages();

<script>
  const iframe = document.getElementById('player-iframe') as HTMLIFrameElement | null;
  const currentLabel = document.getElementById('stream-current');
  const radios = document.querySelectorAll<HTMLInputElement>('input[name="stream"]');

  function selectStream(radio: HTMLInputElement) {
    if (!iframe || !radio) return;

    iframe.src = radio.value;

    if (currentLabel) {
      currentLabel.textContent = `${radio.dataset.quality} · ${radio.dataset.server}`;
    }

    localStorage.setItem('qimochi:stream', radio.id);
  }

  radios.forEach((radio) => {
    radio.addEventListener('change', () => {
      if (radio.checked) selectStream(radio);
    });
  });

  const saved = localStorage.getItem('qimochi:stream');
  if (saved) {
    const savedRadio = document.getElementById(saved) as HTMLInputElement | null;
    if (savedRadio) {
      savedRadio.checked = true;
      selectStream(savedRadio);
    }
  }
</script>

Selamat datang di **Qimochi** — tempat untuk streaming dan download anime batch
dengan pengalaman yang ringan, cepat, dan tanpa ribet.

## Apa Itu Qimochi?

Qimochi adalah koleksi anime yang bisa kamu tonton langsung atau download
dalam berbagai kualitas. Kami fokus pada tiga hal utama:

### 1. Kecepatan

Situs dibangun dengan teknologi statis modern. Setiap halaman dimuat dalam
waktu kurang dari 1 detik, bahkan di koneksi lambat.

### 2. Kemudahan

Navigasi simpel — tidak ada popup, tidak ada iklan mengganggu, tidak ada
redirect aneh. Kamu datang untuk nonton, kami pastikan itu yang kamu dapat.

### 3. Kelengkapan

Koleksi anime terus bertambah. Dari judul klasik sampai musim terbaru,
semuanya ada di satu tempat.

## Fitur Utama

- **Streaming langsung** — tonton tanpa download
- **Download batch** — unduh semua episode sekaligus
- **Multi-kualitas** — 360p, 480p, 720p, 1080p
- **Multiple server** — ganti server kalau yang satu lag
- **Pencarian canggih** — cari dengan typo tetap ketemu
- **Archive lengkap** — browse berdasarkan abjad
- **Genre filter** — temukan sesuai selera

## Cara Menggunakan

### 1. Browse Anime

Buka halaman [Archive](/anime/) untuk melihat semua anime. Atau jelajahi
berdasarkan [genre](/genre/) kalau kamu punya preferensi tertentu.

### 2. Pilih Episode

Setiap anime punya halaman detail dengan daftar episode lengkap. Klik
episode yang mau ditonton.

### 3. Pilih Server & Kualitas

Di halaman tonton, kamu bisa memilih:

- **Kualitas** — dari 360p (hemat kuota) sampai 1080p (kualitas terbaik)
- **Server** — ganti-ganti kalau yang satu bermasalah

Pilihanmu akan **diingat otomatis** untuk kunjungan berikutnya.

### 4. Nonton atau Download

- **Nonton langsung** — klik server, video langsung main
- **Download** — pilih kualitas, klik server download

## Yang Perlu Diketahui

Kami **tidak menyimpan file video** di server kami. Semua konten di-embed
dari layanan pihak ketiga yang tersedia bebas di internet. Kami hanya
menyediakan tautan untuk keperluan referensi.

Untuk informasi lebih lanjut, lihat halaman [DMCA](/dmca/) dan
[Disclaimer](/disclaimer/).

## Bergabung dengan Komunitas

import { getCollection } from 'astro:content';
import type { CollectionEntry } from 'astro:content';

export type BlogEntry = CollectionEntry<'blog'>;

/**
 * Ambil semua artikel blog yang published (bukan draft).
 * Sort by date descending (terbaru dulu).
 */
export async function getPublishedPosts(): Promise<BlogEntry[]> {
  const allPosts = await getCollection('blog', ({ data }) => !data.draft);
  return [...allPosts].sort(
    (a, b) => b.data.date.getTime() - a.data.date.getTime()
  );
}

/**
 * Ambil artikel by slug — return null kalau draft atau tidak ada.
 */
export async function getPublishedPostBySlug(
  slug: string
): Promise<BlogEntry | null> {
  const allPosts = await getCollection('blog');
  const post = allPosts.find(p => p.id === slug);
  if (!post || post.data.draft) return null;
  return post;
}

/**
 * Ambil semua slug artikel published — untuk getStaticPaths.
 */
export async function getAllPostSlugs(): Promise<string[]> {
  const posts = await getPublishedPosts();
  return posts.map(p => p.id);
}

/**
 * Format tanggal ke format Indonesia.
 * Contoh: "15 Februari 2026"
 */
export function formatDate(date: Date): string {
  return new Intl.DateTimeFormat('id-ID', {
    day: 'numeric',
    month: 'long',
    year: 'numeric',
  }).format(date);
}

/**
 * Format tanggal singkat.
 * Contoh: "15 Feb 2026"
 */
export function formatDateShort(date: Date): string {
  return new Intl.DateTimeFormat('id-ID', {
    day: 'numeric',
    month: 'short',
    year: 'numeric',
  }).format(date);
}

/**
 * Ambil artikel terkait berdasarkan tag yang sama.
 * Return max `limit` artikel, exclude artikel itu sendiri.
 */
export function getRelatedPosts(
  current: BlogEntry,
  all: BlogEntry[],
  limit = 3
): BlogEntry[] {
  const currentTags = new Set(current.data.tags);

  if (currentTags.size === 0) return [];

  return all
    .filter(post => post.id !== current.id)
    .map(post => ({
      post,
      score: post.data.tags.filter(t => currentTags.has(t)).length,
    }))
    .filter(({ score }) => score > 0)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit)
    .map(({ post }) => post);
}

/**
 * Ambil semua tag unik dari artikel published + jumlah per tag.
 * Sort by count descending, lalu alfabetis.
 */
export function collectTags(
  posts: BlogEntry[]
): { tag: string; count: number }[] {
  const counts: Record<string, number> = {};

  posts.forEach(post => {
    post.data.tags.forEach(tag => {
      counts[tag] = (counts[tag] || 0) + 1;
    });
  });

  return Object.entries(counts)
    .map(([tag, count]) => ({ tag, count }))
    .sort((a, b) => b.count - a.count || a.tag.localeCompare(b.tag, 'id-ID'));
}

/**
 * Ambil artikel by tag.
 */
export function filterByTag(posts: BlogEntry[], tag: string): BlogEntry[] {
  return posts.filter(post => post.data.tags.includes(tag));
}

/**
 * Ambil artikel by category.
 */
export function filterByCategory(
  posts: BlogEntry[],
  category: string
): BlogEntry[] {
  return posts.filter(post => post.data.category === category);
}

/**
 * Build canonical URL untuk artikel blog.
 */
export function postUrl(slug: string): string {
  return `/blog/${slug}/`;
}

import { getCollection } from 'astro:content';
import { PER_PAGE } from '../constants';
import { sortByRecent, type AnimeEntry } from './anime';

export type { AnimeEntry };

export const STATUS_MAP = {
  ongoing: 'Ongoing',
  complete: 'Completed',
  hiatus: 'Hiatus',
} as const;

export type StatusSlug = keyof typeof STATUS_MAP;

export async function getAnimeByStatus(status: string): Promise<AnimeEntry[]> {
  const allAnime = await getCollection('anime');
  const filtered = allAnime.filter(a => a.data.status === status);
  return sortByRecent(filtered);
}

export async function getStatusData(slug: StatusSlug) {
  const status = STATUS_MAP[slug];
  const anime = await getAnimeByStatus(status);
  const totalPages = Math.ceil(anime.length / PER_PAGE);
  return { status, anime, totalPages };
}

export function buildStatusPaths(
  slug: StatusSlug,
  anime: AnimeEntry[],
  totalPages: number
) {
  const paths = [
    {
      params: { page: undefined },
      props: { anime, totalPages, pageNumber: 1 },
    },
  ];

  for (let page = 2; page <= totalPages; page++) {
    paths.push({
      params: { page: String(page) },
      props: { anime, totalPages, pageNumber: page },
    });
  }

  return paths;
}

export function paginate<T>(items: T[], page: number): T[] {
  const start = (page - 1) * PER_PAGE;
  return items.slice(start, start + PER_PAGE);
}

import { getCollection } from 'astro:content';
import BaseLayout from '~/layouts/BaseLayout.astro';
import Hero from '~/components/Hero.astro';
import Badge from '~/user-components/Badge.astro';
import { URLS, STATUS_VARIANT, type Status } from '~/constants';

const allAnime = await getCollection('anime');

const sorted = [...allAnime].sort((a, b) =>
  a.data.title.localeCompare(b.data.title, 'id-ID')
);

function getFirstLetter(title: string): string {
  const first = title[0]?.toUpperCase() ?? '';
  if (/[A-Z]/.test(first)) return first;
  return '#';
}

const letters = new Set<string>();
sorted.forEach(a => letters.add(getFirstLetter(a.data.title)));

const activeLetters = Array.from(letters).sort((a, b) => {
  if (a === '#') return 1;
  if (b === '#') return -1;
  return a.localeCompare(b);
});

const totalAnime = allAnime.length;

const groups = activeLetters.map(letter => ({
  letter,
  items: sorted.filter(a => getFirstLetter(a.data.title) === letter),
}));

export async function getStaticPaths() {
  const paths = [];

  for (const { genre, slug } of allGenreSlugs()) {
    const { anime, totalPages } = await getGenreData(genre);

    if (anime.length === 0) continue;

    paths.push({
      params: { slug, page: undefined },
      props: { genre, anime, totalPages, pageNumber: 1 },

    for (let page = 2; page <= totalPages; page++) {

        params: { slug, page: String(page) },
        props: { genre, anime, totalPages, pageNumber: page },
      });
    }
  }

  return paths;
}

<script>
  const BATCH_SIZE = 12;

  function normalize(s: string): string {
    return s
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .replace(/[^a-z0-9\s]/g, '')
      .replace(/\s+/g, ' ')
      .trim();
  }

  function levenshtein(a: string, b: string): number {
    if (a === b) return 0;
    if (a.length === 0) return b.length;
    if (b.length === 0) return a.length;

    const matrix: number[][] = [];
    for (let i = 0; i <= b.length; i++) matrix[i] = [i];
    for (let j = 0; j <= a.length; j++) matrix[0][j] = j;

    for (let i = 1; i <= b.length; i++) {
      for (let j = 1; j <= a.length; j++) {
        const cost = a[j - 1] === b[i - 1] ? 0 : 1;
        matrix[i][j] = Math.min(
          matrix[i - 1][j] + 1,
          matrix[i][j - 1] + 1,
          matrix[i - 1][j - 1] + cost
        );
      }
    }

    return matrix[b.length][a.length];
  }

  function scoreTitle(query: string, title: string): number {
    const q = normalize(query);
    const t = normalize(title);
    if (!q || !t) return 0;

    if (t.includes(q)) {
      const pos = t.indexOf(q);
      return 1 - (pos / t.length) * 0.3;
    }

    const words = t.split(' ');
    let best = 0;

    for (const word of words) {
      const dist = levenshtein(q, word);
      const maxLen = Math.max(q.length, word.length);
      const similarity = 1 - dist / maxLen;
      if (similarity > best) best = similarity;
    }

    if (q.length >= 3) {
      for (const word of words) {
        if (word.includes(q) || q.includes(word)) {
          best = Math.max(best, 0.85);
        }
      }
    }

    return best * 0.85;
  }

     ---------------------------------------------------------- */
  const form = document.getElementById('search-form') as HTMLFormElement;

  function render() {
    const visible = matchedCards.slice(0, visibleCount);
    const visibleSet = new Set(visible);

    allCards.forEach(card => {
      card.style.display = visibleSet.has(card) ? '' : 'none';
    });

    const remaining = matchedCards.length - visibleCount;
    const hasMore = remaining > 0;
    loadMoreWrap.hidden = !hasMore;
    loadMoreText.textContent = hasMore
      ? `Muat Lebih Banyak (${remaining} lagi)`
      : 'Muat Lebih Banyak';
  }

  /* ----------------------------------------------------------
     FILTER
     ---------------------------------------------------------- */
  function applyFilter(query: string) {
    const q = query.trim();

    // Empty state — belum ada query
    if (q.length === 0) {
      emptyState.hidden = false;
      status.hidden = true;
      results.hidden = true;
      loadMoreWrap.hidden = true;
      return;
    }

    // Query terlalu pendek
    if (q.length < 2) {
      emptyState.hidden = true;
      status.textContent = 'Ketik minimal 2 karakter.';
      status.hidden = false;
      results.hidden = true;
      loadMoreWrap.hidden = true;
      return;
    }

    // Filter
    const scored = allCards
      .map(card => ({
        card,
        score: scoreTitle(q, card.dataset.animeTitle || ''),
      }))
      .filter(({ score }) => score > 0.4)
      .sort((a, b) => b.score - a.score);

    matchedCards = scored.map(s => s.card);
    visibleCount = BATCH_SIZE;

    emptyState.hidden = true;
    results.hidden = false;

    if (matchedCards.length === 0) {
      status.textContent = `Tidak ada hasil untuk "${q}".`;
      status.hidden = false;
      results.hidden = true;
      loadMoreWrap.hidden = true;
      return;
    }

    status.textContent = `${matchedCards.length} hasil untuk "${q}"`;
    status.hidden = false;
    render();
  }

  /* ----------------------------------------------------------
     EVENTS
     ---------------------------------------------------------- */
  function updateClearBtn() {

  }

    url.searchParams.delete('q');
    window.history.pushState({}, '', url);
  });

    visibleCount += BATCH_SIZE;
    render();
  });

    else url.searchParams.delete('q');

    const p = new URLSearchParams(window.location.search);
    const q = p.get('q') || '';
    input.value = q;
    updateClearBtn();
    applyFilter(q);
  });

  // Initial
  applyFilter(initialQuery);
